cmake_minimum_required(VERSION 3.20)
project(ComicEnhancer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(ComicCore core/src/ComicCore.cpp)
target_include_directories(ComicCore PUBLIC core/include)
target_link_libraries(ComicCore PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ComicCore PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(APPLE)
  target_link_libraries(ComicCore PUBLIC "-framework Accelerate")
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE ComicCore)
add_test(NAME core_tests COMMAND core_tests)
